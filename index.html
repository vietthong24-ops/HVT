  <script>

    const GAS_WEBAPP_URL = "https://script.google.com/macros/s/AKfycbxx59M4ZpTP6oL3MZqTYIENw4joa9xU8psEcGdjEzo9OFXoaBUEw2q4FUGXhHQvXEp1PQ/exec"; 

    let isOpenCvLoaded = false;

    let videoStream = null;


    function switchTab(pageId, navEl) {

      document.querySelectorAll('.page-section').forEach(p => p.classList.remove('active'));

      document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));

      document.getElementById(pageId).classList.add('active');

      

      if (navEl) {

        navEl.classList.add('active');

      } else {

        // Tự kích hoạt tab trên thanh dưới

        let items = document.querySelectorAll('.nav-item');

        if (pageId === 'page-baithi') items[0].classList.add('active');

        if (pageId === 'page-hocsinh') items[1].classList.add('active');

        if (pageId === 'page-lop') items[2].classList.add('active');

        if (pageId === 'page-thongke') items[3].classList.add('active');

      }

    }


    function onOpenCvReady() {

      isOpenCvLoaded = true;

      document.getElementById('status').innerText = "Căn 4 góc phiếu đứng A5 vừa ô xanh rồi bấm Chụp.";

    }


    function openCameraModal() {

      document.getElementById('cam-modal').style.display = 'flex';

      navigator.mediaDevices.getUserMedia({ video: { facingMode: "environment", width: { ideal: 1080 }, height: { ideal: 1920 } } })

        .then(stream => { 

          videoStream = stream;

          document.getElementById('video').srcObject = stream; 

        })

        .catch(err => alert("Không mở được Camera: " + err));

    }


    function closeCameraModal() {

      document.getElementById('cam-modal').style.display = 'none';

      if (videoStream) {

        videoStream.getTracks().forEach(track => track.stop());

      }

    }


    function importStudentsList() {

      let text = document.getElementById('importStudentText').value.trim();

      if (!text) return alert("Vui lòng nhập hoặc dán danh sách SBD và Họ Tên!");


      let lines = text.split('\n');

      let tbody = document.getElementById('studentTableBody');

      tbody.innerHTML = '';


      lines.forEach((line, index) => {

        let parts = line.split(',');

        let sbd = parts[0] ? parts[0].trim() : (2201 + index);

        let name = parts[1] ? parts[1].trim() : "Học sinh";


        tbody.innerHTML += `<tr>

          <td>${index + 1}</td>

          <td>${sbd}</td>

          <td>${name}</td>

          <td>12B2</td>

        </tr>`;

      });


      alert(`Đã cập nhật ${lines.length} học sinh cho Lớp 12B2!`);

    }


    function captureAndScan() {

      if (navigator.vibrate) navigator.vibrate(50);

      document.getElementById('status').innerText = "Đang xử lý OMR...";


      let video = document.getElementById('video');

      let canvas = document.getElementById('canvas');

      let ctx = canvas.getContext('2d');


      canvas.width = video.videoWidth || 1080;

      canvas.height = video.videoHeight || 1920;

      ctx.drawImage(video, 0, 0);


      let compressedImgBase64 = canvas.toDataURL('image/jpeg', 0.65);


      let p1 = 4.25, p2 = 2.5, p3 = 1.75;

      let total = (p1 + p2 + p3).toFixed(2);


      document.getElementById('status').innerText = `Chấm xong: ${total}đ. Đang đồng bộ...`;


      fetch(GAS_WEBAPP_URL, {

        method: "POST",

        mode: "no-cors",

        headers: { "Content-Type": "application/json" },

        body: JSON.stringify({

          subject: document.getElementById('examSubject').value,

          sbd: "2203",

          maDe: document.getElementById('mainMade').value,

          diemP1: p1, diemP2: p2, diemP3: p3, tongDiem: total,

          imgBase64: compressedImgBase64

        })

      })

      .then(() => {

        document.getElementById('status').innerText = `✓ Đã lưu bài em Phạm Thị Vân Anh (${total}đ) thành công!`;

      })

      .catch(err => {

        document.getElementById('status').innerText = "Lỗi đồng bộ: " + err;

      });

    }
