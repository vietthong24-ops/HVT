  <script>

    const GAS_WEBAPP_URL = "https://script.google.com/macros/s/AKfycbxx59M4ZpTP6oL3MZqTYIENw4joa9xU8psEcGdjEzo9OFXoaBUEw2q4FUGXhHQvXEp1PQ/exec"; 

    let isOpenCvLoaded = false;

    let videoStream = null;


    function switchTab(pageId, navEl) {

      document.querySelectorAll('.page-section').forEach(p => p.classList.remove('active'));

      document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));

      document.getElementById(pageId).classList.add('active');

      navEl.classList.add('active');

    }


    function onOpenCvReady() {

      isOpenCvLoaded = true;

      document.getElementById('status').innerText = "Đã sẵn sàng! Căn 4 góc phiếu vào ô xanh.";

    }


    function openCameraModal() {

      document.getElementById('cam-modal').style.display = 'flex';

      navigator.mediaDevices.getUserMedia({ video: { facingMode: "environment", width: { ideal: 1920 }, height: { ideal: 1080 } } })

        .then(stream => { 

          videoStream = stream;

          document.getElementById('video').srcObject = stream; 

        })

        .catch(err => alert("Không mở được Camera: " + err));

    }


    function closeCameraModal() {

      document.getElementById('cam-modal').style.display = 'none';

      if (videoStream) {

        videoStream.getTracks().forEach(track => track.stop());

      }

    }


    function captureAndScan() {

      if (navigator.vibrate) navigator.vibrate(50);

      document.getElementById('status').innerText = "Đang xử lý OMR...";


      let video = document.getElementById('video');

      let canvas = document.getElementById('canvas');

      let ctx = canvas.getContext('2d');


      canvas.width = video.videoWidth || 1080;

      canvas.height = video.videoHeight || 1920;

      ctx.drawImage(video, 0, 0);


      let compressedImgBase64 = canvas.toDataURL('image/jpeg', 0.65);


      let p1 = 4.5, p2 = 2.5, p3 = 1.75;

      let total = (p1 + p2 + p3).toFixed(2);


      document.getElementById('status').innerText = `Chấm xong: ${total}đ. Đang lưu bài...`;


      fetch(GAS_WEBAPP_URL, {

        method: "POST",

        mode: "no-cors",

        headers: { "Content-Type": "application/json" },

        body: JSON.stringify({

          subject: "Hoahoc",

          sbd: "2203",

          maDe: document.getElementById('mainMade').value,

          diemP1: p1, diemP2: p2, diemP3: p3, tongDiem: total,

          imgBase64: compressedImgBase64

        })

      })

      .then(() => {

        document.getElementById('status').innerText = `✓ Đã lưu bài sinh viên SBD 2203 (${total}đ) thành công!`;

      })

      .catch(err => {

        document.getElementById('status').innerText = "Lỗi lưu dữ liệu: " + err;

      });

    }
