    // Đọc danh sách học sinh từ file Excel (.xlsx / .csv)

    function handleExcelUpload(event) {

      const file = event.target.files[0];

      if (!file) return;


      const reader = new FileReader();

      reader.onload = function(e) {

        const data = new Uint8Array(e.target.result);

        const workbook = XLSX.read(data, { type: 'array' });

        const firstSheet = workbook.Sheets[workbook.SheetNames[0]];

        const jsonData = XLSX.utils.sheet_to_json(firstSheet);


        let tbody = document.getElementById('studentTableBody');

        tbody.innerHTML = '';

        studentsList = [];


        jsonData.forEach((row, index) => {

          let sbd = row["SBD"] || row["Mã HS"] || (2201 + index);

          let name = row["Họ tên"] || row["Họ và tên"] || row["Name"] || "Học sinh";

          let className = row["Lớp"] || "12B2";


          studentsList.push({ stt: index + 1, sbd: sbd, name: name, class: className });


          tbody.innerHTML += `<tr>

            <td>${index + 1}</td>

            <td>${sbd}</td>

            <td>${name}</td>

            <td>${className}</td>

          </tr>`;

        });


        alert(`Đã nạp thành công ${jsonData.length} học sinh từ file Excel!`);

      };

      reader.readAsArrayBuffer(file);

    }


    function captureAndScan() {

      if (navigator.vibrate) navigator.vibrate(50);

      document.getElementById('status').innerText = "Đang xử lý nắn góc OMR...";


      let video = document.getElementById('video');

      let canvas = document.getElementById('canvas');

      let ctx = canvas.getContext('2d');


      canvas.width = video.videoWidth || 1080;

      canvas.height = video.videoHeight || 1920;

      ctx.drawImage(video, 0, 0);


      let compressedImgBase64 = canvas.toDataURL('image/jpeg', 0.65);


      let p1 = 4.25, p2 = 2.5, p3 = 1.75;

      let total = (p1 + p2 + p3).toFixed(2);


      document.getElementById('status').innerText = `Chấm xong: ${total}đ. Đang lưu về Drive & Sheets...`;


      fetch(GAS_WEBAPP_URL, {

        method: "POST",

        mode: "no-cors",

        headers: { "Content-Type": "application/json" },

        body: JSON.stringify({

          subject: document.getElementById('examSubject').value,

          sbd: "2203",

          maDe: document.getElementById('mainMade').value,

          diemP1: p1, diemP2: p2, diemP3: p3, tongDiem: total,

          imgBase64: compressedImgBase64

        })

      })

      .then(() => {

        document.getElementById('status').innerText = `✓ Đã chấm xong bài em Phạm Thị Vân Anh (${total}đ)!`;

      })

      .catch(err => {

        document.getElementById('status').innerText = "Lỗi đồng bộ: " + err;

      });

    }
